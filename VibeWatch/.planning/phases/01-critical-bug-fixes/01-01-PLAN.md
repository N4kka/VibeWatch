    import XCTest
    @testable import VibeWatchApp

    final class AnalyticsInsightsTests: XCTestCase {
        let userId = "test-user-mood-\(UUID().uuidString)"

        override func setUp() async throws {
            // Clean any prior test data
            SQLiteService.shared.execute(
                "DELETE FROM user_clip_history WHERE user_id = ?",
                parameters: [userId]
            )
        }

        override func tearDown() async throws {
            SQLiteService.shared.execute(
                "DELETE FROM user_clip_history WHERE user_id = ?",
                parameters: [userId]
            )
        }

        func testGenerateUserStatisticsMoodAnalysisIsNotNil() async throws {
            // BUG-04: moodAnalysis must not be nil after fix
            let stats = await AnalyticsInsightsService.shared.generateUserStatistics(
                userId: userId,
                timeframe: .allTime
            )
            XCTAssertNotNil(stats.moodAnalysis,
                "generateUserStatistics must return non-nil moodAnalysis after BUG-04 fix")
        }

        func testMoodDistributionReflectsGenreHistory() async throws {
            // BUG-04: Seed 5 comedy + 5 action items, expect Light and Intense in distribution
            let db = SQLiteService.shared
            for i in 0..<5 {
                db.execute("""
                    INSERT INTO user_clip_history (id, user_id, media_id, genre_ids, watched_at)
                    VALUES (?, ?, ?, ?, datetime('now'))
                """, parameters: [
                    "test-comedy-\(i)-\(userId)",
                    userId,
                    "movie-comedy-\(i)",
                    "35"  // Comedy → Light
                ])
            }
            for i in 0..<5 {
                db.execute("""
                    INSERT INTO user_clip_history (id, user_id, media_id, genre_ids, watched_at)
                    VALUES (?, ?, ?, ?, datetime('now'))
                """, parameters: [
                    "test-action-\(i)-\(userId)",
                    userId,
                    "movie-action-\(i)",
                    "28"  // Action → Intense
                ])
            }

            let stats = await AnalyticsInsightsService.shared.generateUserStatistics(
                userId: userId,
                timeframe: .allTime
            )
            let dist = stats.moodAnalysis?.moodDistribution ?? [:]
            XCTAssertNotNil(dist["Light"], "Light mood expected from comedy genre IDs")
            XCTAssertNotNil(dist["Intense"], "Intense mood expected from action genre IDs")
        }

        func testMoodDistributionIsEmptyWhenInsufficientHistory() async throws {
            // BUG-04: Fewer than 5 items → empty moodDistribution but non-nil moodAnalysis
            // Seed only 2 rows
            let db = SQLiteService.shared
            for i in 0..<2 {
                db.execute("""
                    INSERT INTO user_clip_history (id, user_id, media_id, genre_ids, watched_at)
                    VALUES (?, ?, ?, ?, datetime('now'))
                """, parameters: [
                    "test-few-\(i)-\(userId)", userId, "movie-\(i)", "35"

            )

                "moodAnalysis must be non-nil even with insufficient history")

                "moodDistribution must be empty when fewer than 5 history items")
        }
    }
